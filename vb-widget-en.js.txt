// VB · Production widget «Veseli bobry» (English)
// Scriptable → new script → paste this text → name it «VB».
// Data: widget.json from the site, fallback — GitHub (updated together with the site). Version 2 EN.
// Sizes: small (3 metrics) and medium (5 metrics).

const SITE = "https://veselibobry.netlify.app/";
// Data sources in order: site → GitHub (if Netlify is behind or unavailable)
const SOURCES = [
  SITE + "widget.json",
  "https://raw.githubusercontent.com/Slavaberin/Veselibobry/main/widget.json",
];
const C = {
  bg: new Color("#1a1a0f"),
  paper: new Color("#f5f0e8"),
  muted: new Color("#f5f0e8", 0.5),
  amber: new Color("#c17f24"),
  good: new Color("#5fa046"),
  warn: new Color("#c17f24"),
  bad: new Color("#d0503c"),
};

// ── data: network; offline → last saved copy ──
const fm = FileManager.local();
const cachePath = fm.joinPath(fm.documentsDirectory(), "vb-widget-cache.json");
let d = null, offline = false, errs = [];
for (const url of SOURCES) {
  try {
    const req = new Request(url + "?t=" + Date.now());
    req.timeoutInterval = 15;
    const txt = await req.loadString();
    const code = req.response ? req.response.statusCode : 200;
    if (code !== 200) throw new Error("HTTP " + code);
    d = JSON.parse(txt);
    fm.writeString(cachePath, JSON.stringify(d));
    break;
  } catch (e) {
    errs.push((url.includes("github") ? "GitHub: " : "Site: ") + String(e.message || e).slice(0, 60));
  }
}
if (!d) {
  offline = true;
  if (fm.fileExists(cachePath)) d = JSON.parse(fm.readString(cachePath));
}

const fmt = n => (n == null ? "—" : Math.round(n).toLocaleString("en-US"));
const family = config.widgetFamily || "medium";
const w = new ListWidget();
w.backgroundColor = C.bg;
w.url = SITE + "?lang=en";
w.setPadding(14, 16, 12, 16);
w.refreshAfterDate = new Date(Date.now() + 15 * 60 * 1000);

// ── header: VB · date ──
const head = w.addStack();
head.centerAlignContent();
const v = head.addText("V"); v.font = Font.boldSystemFont(16); v.textColor = C.paper;
const b = head.addText("B"); b.font = Font.boldSystemFont(16); b.textColor = C.amber;
head.addSpacer(6);
const hd = head.addText(d && d.prod_last ? d.prod_last.date : "");
hd.font = Font.mediumMonospacedSystemFont(11); hd.textColor = C.muted;
head.addSpacer();
if (family !== "small" && d) {
  const up = head.addText((offline ? "offline · " : "") + (d.updated || "").slice(11));
  up.font = Font.mediumMonospacedSystemFont(10); up.textColor = offline ? C.bad : C.muted;
}
w.addSpacer(family === "small" ? 6 : 8);

if (!d) {
  const t = w.addText("No data.");
  t.font = Font.systemFont(12); t.textColor = C.paper;
  const e = w.addText(errs.join("\n"));
  e.font = Font.mediumMonospacedSystemFont(9); e.textColor = C.muted;
} else {
  const rows = [];
  rows.push({ label: family === "small" ? "Produced" : "Produced (" + (d.prod_last ? d.prod_last.date : "—") + ")",
              value: d.prod_last ? fmt(d.prod_last.value) : "—", unit: "coils" });
  if (family !== "small") rows.push({ label: "This month", value: fmt(d.prod_month), unit: "coils" });
  rows.push({ label: family === "small" ? "Stock" : "Total coil stock", value: d.stock ? fmt(d.stock.value) : "—", unit: "" });
  rows.push({ label: family === "small" ? "Stearic" : "Stearic share",
              value: d.stearic ? d.stearic.value.toFixed(1) : "—", unit: "%", status: d.stearic && d.stearic.status });
  if (family !== "small")
    rows.push({ label: "QM defects", value: d.qm ? d.qm.value.toFixed(1) : "—", unit: "%", status: d.qm && d.qm.status });

  const big = family === "small";
  rows.forEach((r, i) => {
    const s = w.addStack();
    s.centerAlignContent();
    const l = s.addText(r.label);
    l.font = Font.systemFont(big ? 11 : 12); l.textColor = C.muted; l.lineLimit = 1;
    s.addSpacer();
    const val = s.addText(r.value);
    val.font = Font.boldMonospacedSystemFont(big ? 14 : 13);
    val.textColor = i === 0 ? C.amber : C.paper;
    if (r.unit) {
      s.addSpacer(2);
      const u = s.addText(r.unit); u.font = Font.mediumMonospacedSystemFont(10); u.textColor = C.muted;
    }
    s.addSpacer(4);
    const dot = s.addText(r.status ? "●" : " ");
    dot.font = Font.systemFont(9);
    dot.textColor = r.status === "good" ? C.good : r.status === "warn" ? C.warn : r.status === "bad" ? C.bad : C.bg;
    if (i < rows.length - 1) w.addSpacer(big ? 5 : 3);
  });
}
w.addSpacer();

if (config.runsInWidget) {
  Script.setWidget(w);
} else {
  await w.presentMedium();
}
Script.complete();
